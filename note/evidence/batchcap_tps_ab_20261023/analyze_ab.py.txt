#!/usr/bin/env python3
"""Analysis for batchcap_tps_ab_20261023: reads ONLY the raw per-run JSON reports (measurement block); stats from stats_util.py."""
import csv, json, math, os, sys
sys.path.insert(0, "/home/abc/chain-n/metanode/execution/scripts/test/evidence")
from stats_util import mean, sample_sd, t_two_tailed_p, t_critical

D = "/home/abc/chain-n/metanode/note/evidence/batchcap_tps_ab_20261023/raft_ab_1m"
order = [l.split() for l in open(f"{D}/order.txt") if l.strip()]
rows, gate = [], []
for pair, v in order:
    m = json.load(open(f"{D}/ab_{v}_p{pair}_run1.json"))["measurement"]
    rcv = sum((m.get("received_by_validator") or {}).values())
    ok = (m["loss_vs_sent"] == 0 and m["bytes_sent_txs"] == m["committed_ours"] and str(m["parity_status"]).startswith("OK:"))
    gate.append((pair, v, ok, m["loss_vs_sent"], m["parity_status"]))
    rows.append({"pair": int(pair), "variant": v, "bytes_sent_txs": m["bytes_sent_txs"], "committed_ours": m["committed_ours"], "received_sum": rcv,
                 "loss_vs_sent": m["loss_vs_sent"], "tps_end_to_end": m["tps_end_to_end"], "tps_during_injection": m["tps_during_injection"],
                 "tps_active_commit_window": m["tps_active_commit_window"], "inject_seconds": m["inject_seconds"],
                 "end_to_end_seconds": m["end_to_end_seconds"], "parity_status": m["parity_status"]})
with open(f"{D}/summary.csv", "w", newline="") as f:
    w = csv.DictWriter(f, fieldnames=list(rows[0].keys())); w.writeheader(); w.writerows(sorted(rows, key=lambda r: (r["pair"], r["variant"])))

def paired(metric):
    by = {(r["pair"], r["variant"]): r[metric] for r in rows}
    pairs = sorted({r["pair"] for r in rows})
    A_ = [by[(p, "A")] for p in pairs]; B_ = [by[(p, "B")] for p in pairs]
    d = [n - p for p, n in zip(A_, B_)]; pct = [100.0 * (n - p) / p for p, n in zip(A_, B_)]
    n = len(d); md, sd = mean(d), sample_sd(d)
    t = md / (sd / math.sqrt(n)) if sd > 0 else float("inf")
    return {"n_pairs": n, "mean_A": mean(A_), "sd_A": sample_sd(A_), "mean_B": mean(B_), "sd_B": sample_sd(B_),
            "mean_diff": md, "sd_diff": sd, "mean_diff_pct": mean(pct), "sd_diff_pct": sample_sd(pct), "diff_pct_per_pair": pct,
            "t": t, "df": n - 1, "p_two_tailed": t_two_tailed_p(abs(t), n - 1)}

res = {"integrity_gate_all_pass": all(g[2] for g in gate), "runs": len(rows), "metrics": {k: paired(k) for k in
       ("tps_end_to_end", "tps_during_injection", "tps_active_commit_window", "end_to_end_seconds")}}
p = res["metrics"]["tps_end_to_end"]
res["verdict_rule_applied"] = ("degradation" if p["mean_diff_pct"] <= -3.0 and p["p_two_tailed"] < 0.05 else
                               "improvement" if p["mean_diff_pct"] >= 3.0 and p["p_two_tailed"] < 0.05 else "no_significant_difference")
_m = res["metrics"]["tps_end_to_end"]
_tc = t_critical(_m["df"]); _half = _tc * _m["sd_diff_pct"] / math.sqrt(_m["n_pairs"])
res["tps_end_to_end_diff_pct_ci95"] = {"t_critical": _tc, "low": _m["mean_diff_pct"] - _half, "high": _m["mean_diff_pct"] + _half,
                                       "formula": "mean_diff_pct +- t_critical(df,0.05)*sd_diff_pct/sqrt(n)"}
print(f"95% CI of mean diff (end_to_end, %): [{res['tps_end_to_end_diff_pct_ci95']['low']:+.2f}, {res['tps_end_to_end_diff_pct_ci95']['high']:+.2f}]")
json.dump(res, open(f"{D}/stats_summary.json", "w"), indent=2)
print(json.dumps({"integrity_gate_all_pass": res["integrity_gate_all_pass"], "runs": res["runs"], "verdict": res["verdict_rule_applied"]}))
for k, v in res["metrics"].items():
    print(f"{k}: A {v['mean_A']:.2f}±{v['sd_A']:.2f}  B {v['mean_B']:.2f}±{v['sd_B']:.2f}  diff {v['mean_diff']:+.2f} ({v['mean_diff_pct']:+.2f}%, sd {v['sd_diff_pct']:.2f}%)  t={v['t']:.3f} df={v['df']} p={v['p_two_tailed']:.4f}")
print("per-pair % (B vs A, end_to_end):", [round(x, 2) for x in p["diff_pct_per_pair"]])
for g in gate: print(g)

# manipulation check (preregistered): mean txs/applied entry on n0, B must exceed A by >= 25%
import re
def txs_per_entry(v, pair):
    for l in open(f"{D}/ab_{v}_p{pair}_metrics.txt"):
        if l.startswith("n0 "):
            m = re.search(r"applied=(\d+) txs=(\d+)", l)
            return int(m.group(2)) / int(m.group(1))
tpe = {v: [txs_per_entry(v, p) for p in range(1, 11)] for v in ("A", "B")}
ra, rb = mean(tpe["A"]), mean(tpe["B"])
chk = {"txs_per_entry_A": ra, "txs_per_entry_B": rb, "ratio_B_over_A": rb / ra, "passes_25pct": rb >= 1.25 * ra, "A_each": tpe["A"], "B_each": tpe["B"]}
res["manipulation_check"] = chk
json.dump(res, open(f"{D}/stats_summary.json", "w"), indent=2)
print("manipulation check:", {k: (round(v, 2) if isinstance(v, float) else v) for k, v in chk.items() if k not in ("A_each", "B_each")})
print("txs/entry A:", [round(x) for x in tpe["A"]]); print("txs/entry B:", [round(x) for x in tpe["B"]])
