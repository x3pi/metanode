//! Minimal lost-wake-up reproduction attempt modelled on NOMT's IoHandle usage.
//!
//! Pattern (consensus/vendor/nomt/nomt/src/io/mod.rs): a committing thread submits `EXPECTED` commands to I/O worker
//! threads over one channel and then calls `recv()` on a per-handle completion channel exactly `EXPECTED` times; the
//! workers send each completion with `completion_sender.send(..)`. A hang was observed with 11 completions queued and the
//! receiver parked forever.
//!
//! This program runs that pattern for a configurable time under CPU contention and reports a HANG (no progress for
//! STUCK_SECS) together with the receiver's queue length: len() > 0 while stuck == lost wake-up.
//! It prints raw counters only; "no hang" means "not reproduced in this scope", nothing more.
use std::sync::atomic::{AtomicBool, AtomicU64, Ordering};
use std::sync::Arc;
use std::thread;
use std::time::{Duration, Instant};

const EXPECTED: usize = 11;

fn env_u64(k: &str, d: u64) -> u64 {
    std::env::var(k).ok().and_then(|v| v.parse().ok()).unwrap_or(d)
}

trait Chan<T: Send + 'static>: Send + Sync {
    fn send(&self, v: T);
    fn recv(&self) -> T;
    fn len(&self) -> usize;
}

// ---- crossbeam unbounded -------------------------------------------------------------------------------------------
struct Cb<T> {
    tx: crossbeam_channel::Sender<T>,
    rx: crossbeam_channel::Receiver<T>,
}
impl<T: Send + 'static> Chan<T> for Cb<T> {
    fn send(&self, v: T) { self.tx.send(v).unwrap() }
    fn recv(&self) -> T { self.rx.recv().unwrap() }
    fn len(&self) -> usize { self.rx.len() }
}

// ---- std mpsc (Receiver is not Sync: guard with a mutex; len tracked by a counter) -------------------------------------
struct Std<T> {
    tx: std::sync::mpsc::Sender<T>,
    rx: std::sync::Mutex<std::sync::mpsc::Receiver<T>>,
    queued: AtomicU64,
}
impl<T: Send + 'static> Chan<T> for Std<T> {
    fn send(&self, v: T) { self.queued.fetch_add(1, Ordering::SeqCst); self.tx.send(v).unwrap() }
    fn recv(&self) -> T { let r = self.rx.lock().unwrap().recv().unwrap(); self.queued.fetch_sub(1, Ordering::SeqCst); r }
    fn len(&self) -> usize { self.queued.load(Ordering::SeqCst) as usize }
}

fn make<T: Send + 'static>(kind: &str) -> Arc<dyn Chan<T>> {
    match kind {
        "std" => { let (tx, rx) = std::sync::mpsc::channel(); Arc::new(Std { tx, rx: std::sync::Mutex::new(rx), queued: AtomicU64::new(0) }) }
        _ => { let (tx, rx) = crossbeam_channel::unbounded(); Arc::new(Cb { tx, rx }) }
    }
}

fn main() {
    let kind = std::env::var("CH").unwrap_or_else(|_| "crossbeam".into());
    let secs = env_u64("RUN_SECS", 60);
    let instances = env_u64("INSTANCES", 3) as usize; // NOMT had 3 instances
    let workers = env_u64("WORKERS", 4) as usize;
    let load = env_u64("LOAD_THREADS", 0) as usize; // CPU-burning threads
    let stuck_secs = env_u64("STUCK_SECS", 5);
    let jitter_us = env_u64("JITTER_US", 50);
    println!("lw_repro kind={kind} run_secs={secs} instances={instances} workers/instance={workers} load_threads={load} stuck_secs={stuck_secs} jitter_us={jitter_us} cpus={}", thread::available_parallelism().map(|n| n.get()).unwrap_or(0));

    let stop = Arc::new(AtomicBool::new(false));
    let mut load_handles = vec![];
    for _ in 0..load {
        let stop = stop.clone();
        load_handles.push(thread::spawn(move || { let mut x = 1u64; while !stop.load(Ordering::Relaxed) { x = x.wrapping_mul(6364136223846793005).wrapping_add(1442695040888963407); std::hint::black_box(x); } }));
    }

    let progress: Vec<Arc<AtomicU64>> = (0..instances).map(|_| Arc::new(AtomicU64::new(0))).collect();
    let lens: Vec<Arc<std::sync::Mutex<Option<Arc<dyn Chan<u64>>>>>> = (0..instances).map(|_| Arc::new(std::sync::Mutex::new(None))).collect();
    let mut handles = vec![];
    for i in 0..instances {
        let progress = progress[i].clone();
        let lenslot = lens[i].clone();
        let stop = stop.clone();
        let kind = kind.clone();
        handles.push(thread::spawn(move || {
            let cmds: Arc<dyn Chan<u64>> = make(&kind);
            let comps: Arc<dyn Chan<u64>> = make(&kind);
            *lenslot.lock().unwrap() = Some(comps.clone());
            for w in 0..workers {
                let cmds = cmds.clone();
                let comps = comps.clone();
                let stop = stop.clone();
                thread::spawn(move || {
                    let mut seed = (w as u64 + 1) * 0x9E3779B97F4A7C15;
                    while !stop.load(Ordering::Relaxed) {
                        let c = cmds.recv();
                        if c == u64::MAX { break; }
                        seed ^= seed << 13; seed ^= seed >> 7; seed ^= seed << 17;
                        // emulate variable I/O latency: sometimes none, sometimes a short sleep or yield
                        match seed % 4 { 0 => {}, 1 => thread::yield_now(), 2 => thread::sleep(Duration::from_micros(seed % (jitter_us + 1))), _ => { for _ in 0..(seed % 200) { std::hint::spin_loop(); } } }
                        comps.send(c);
                    }
                });
            }
            let mut round = 0u64;
            while !stop.load(Ordering::Relaxed) {
                for k in 0..EXPECTED { cmds.send(round * 100 + k as u64); }
                for _ in 0..EXPECTED { let _ = comps.recv(); }
                round += 1;
                progress.store(round, Ordering::Relaxed);
            }
            for _ in 0..workers { cmds.send(u64::MAX); }
        }));
    }

    let start = Instant::now();
    let mut last = vec![0u64; instances];
    let mut last_change = vec![Instant::now(); instances];
    let mut hung = 0;
    while start.elapsed() < Duration::from_secs(secs) {
        thread::sleep(Duration::from_millis(500));
        for i in 0..instances {
            let p = progress[i].load(Ordering::Relaxed);
            if p != last[i] { last[i] = p; last_change[i] = Instant::now(); }
            else if last_change[i].elapsed() > Duration::from_secs(stuck_secs) {
                let qlen = lens[i].lock().unwrap().as_ref().map(|c| c.len()).unwrap_or(0);
                println!("HANG instance={i} no progress for {:?} at round={p} receiver_queue_len={qlen}  => {}", last_change[i].elapsed(), if qlen > 0 { "LOST WAKE-UP (queued messages but receiver not woken)" } else { "stall with empty queue (not the NOMT signature)" });
                hung += 1;
                last_change[i] = Instant::now();
            }
        }
    }
    stop.store(true, Ordering::Relaxed);
    let total: u64 = progress.iter().map(|p| p.load(Ordering::Relaxed)).sum();
    println!("RESULT kind={kind} elapsed={:?} total_rounds={total} total_recv_calls={} hangs_observed={hung}", start.elapsed(), total * EXPECTED as u64);
    std::process::exit(if hung > 0 { 2 } else { 0 });
}
